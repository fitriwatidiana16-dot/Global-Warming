        function init3DEarthCover() {
            const container = document.getElementById('canvas-earth-cover');
            if (!container) return;
            container.innerHTML = '';

            const width = container.clientWidth || 300;
            const height = container.clientHeight || 300;

            const scene = new THREE.Scene();
            const camera = new THREE.PerspectiveCamera(45, width / height, 0.1, 1000);
            camera.position.z = 2.5;

            const renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
            renderer.setSize(width, height);
            renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
            container.appendChild(renderer.domElement);

            const geometry = new THREE.SphereGeometry(0.85, 32, 32);
            const count = geometry.attributes.position.count;
            const colors = [];
            const colorHealthy = new THREE.Color(0x0284c7);
            const colorWarm = new THREE.Color(0xea580c);

            for (let i = 0; i < count; i++) {
                const x = geometry.attributes.position.getX(i);
                if (x < 0) {
                    colors.push(colorHealthy.r, colorHealthy.g, colorHealthy.b);
                } else {
                    colors.push(colorWarm.r, colorWarm.g, colorWarm.b);
                }
            }
            geometry.setAttribute('color', new THREE.Float32BufferAttribute(colors, 3));

            const material = new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.5, metalness: 0.1 });
            const earth = new THREE.Mesh(geometry, material);
            scene.add(earth);

            const atmoGeom = new THREE.SphereGeometry(0.98, 32, 32);
            const atmoMat = new THREE.MeshBasicMaterial({ color: 0x38bdf8, transparent: true, opacity: 0.25, wireframe: true });
            const atmosphere = new THREE.Mesh(atmoGeom, atmoMat);
            scene.add(atmosphere);

            const ambientLight = new THREE.AmbientLight(0xffffff, 0.9);
            scene.add(ambientLight);
            const dirLight = new THREE.DirectionalLight(0xffffff, 1.2);
            dirLight.position.set(5, 3, 5);
            scene.add(dirLight);

            function animate() {
                requestAnimationFrame(animate);
                earth.rotation.y += 0.005;
                atmosphere.rotation.y += 0.003;
                renderer.render(scene, camera);
            }
            animate();
        }

        // REAL 3D ATMOSPHERE & SOLAR PHYSICS SIMULATOR
        // ==========================================
        let sim3DScene, sim3DCamera, sim3DRenderer, sim3DEarth, sim3DSun, sim3DAtmoMesh;
        let solarRaysGroup, infraRaysGroup, gasParticlesGroup;

        function createProceduralEarthTexture() {
            const canvas = document.createElement('canvas');
            canvas.width = 512;
            canvas.height = 256;
            const ctx = canvas.getContext('2d');

            ctx.fillStyle = '#0f4c81';
            ctx.fillRect(0, 0, 512, 256);

            ctx.fillStyle = '#2e7d32';
            for (let i = 0; i < 18; i++) {
                ctx.beginPath();
                const x = Math.random() * 512;
                const y = 40 + Math.random() * 176;
                const r = 25 + Math.random() * 45;
                ctx.arc(x, y, r, 0, Math.PI * 2);
                ctx.fill();
            }

            ctx.fillStyle = '#f8fafc';
            ctx.fillRect(0, 0, 512, 18);
            ctx.fillRect(0, 238, 512, 18);

            return new THREE.CanvasTexture(canvas);
        }

        function initReal3DSimulator() {
            const container = document.getElementById('canvas-3d-sim');
            if (!container) return;
            container.innerHTML = '';

            const width = container.clientWidth || 600;
            const height = container.clientHeight || 380;

            sim3DScene = new THREE.Scene();
            sim3DScene.background = new THREE.Color(0x0a0f1d);

            sim3DCamera = new THREE.PerspectiveCamera(45, width / height, 0.1, 1000);
            sim3DCamera.position.set(0, 1.2, 4.2);

            sim3DRenderer = new THREE.WebGLRenderer({ antialias: true, alpha: false });
            sim3DRenderer.setSize(width, height);
            sim3DRenderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
            container.appendChild(sim3DRenderer.domElement);

            let controls;
            if (typeof THREE.OrbitControls !== 'undefined') {
                controls = new THREE.OrbitControls(sim3DCamera, sim3DRenderer.domElement);
                controls.enableDamping = true;
                controls.dampingFactor = 0.05;
                controls.maxDistance = 8;
                controls.minDistance = 2;
            }

            const earthTexture = createProceduralEarthTexture();
            const earthGeom = new THREE.SphereGeometry(1, 32, 32);
            const earthMat = new THREE.MeshStandardMaterial({
                map: earthTexture,
                roughness: 0.6,
                metalness: 0.1
            });
            sim3DEarth = new THREE.Mesh(earthGeom, earthMat);
            sim3DScene.add(sim3DEarth);

            const atmoGeom = new THREE.SphereGeometry(1.28, 32, 32);
            const atmoMat = new THREE.MeshBasicMaterial({
                color: 0x38bdf8,
                transparent: true,
                opacity: 0.22,
                wireframe: true
            });
            sim3DAtmoMesh = new THREE.Mesh(atmoGeom, atmoMat);
            sim3DScene.add(sim3DAtmoMesh);

            const sunGeom = new THREE.SphereGeometry(0.38, 24, 24);
            const sunMat = new THREE.MeshBasicMaterial({ color: 0xfacc15 });
            sim3DSun = new THREE.Mesh(sunGeom, sunMat);
            sim3DSun.position.set(-3.6, 1.8, -1.2);
            sim3DScene.add(sim3DSun);

            const sunLight = new THREE.PointLight(0xfff5cc, 2.2, 100);
            sunLight.position.copy(sim3DSun.position);
            sim3DScene.add(sunLight);

            const ambientLight = new THREE.AmbientLight(0x334155, 1.5);
            sim3DScene.add(ambientLight);

            solarRaysGroup = new THREE.Group();
            infraRaysGroup = new THREE.Group();
            gasParticlesGroup = new THREE.Group();

            sim3DScene.add(solarRaysGroup);
            sim3DScene.add(infraRaysGroup);
            sim3DScene.add(gasParticlesGroup);

            update3DSimulation();

            window.addEventListener('resize', () => {
                if (!container) return;
                const w = container.clientWidth;
                const h = container.clientHeight;
                if (w && h && sim3DCamera && sim3DRenderer) {
                    sim3DCamera.aspect = w / h;
                    sim3DCamera.updateProjectionMatrix();
                    sim3DRenderer.setSize(w, h);
                }
            });

            function animate3DSim() {
                requestAnimationFrame(animate3DSim);
                if (controls) controls.update();

                if (sim3DEarth) sim3DEarth.rotation.y += 0.003;
                if (gasParticlesGroup) gasParticlesGroup.rotation.y += 0.002;

                if (solarRaysGroup) {
                    solarRaysGroup.children.forEach(ray => {
                        ray.position.x += ray.userData.vx;
                        ray.position.y += ray.userData.vy;
                        ray.position.z += ray.userData.vz;

                        if (ray.position.distanceTo(sim3DEarth.position) < 1.0) {
                            ray.position.copy(sim3DSun.position);
                        }
                    });
                }

                if (infraRaysGroup) {
                    infraRaysGroup.children.forEach(ray => {
                        ray.position.x += ray.userData.vx;
                        ray.position.y += ray.userData.vy;
                        ray.position.z += ray.userData.vz;

                        if (ray.position.distanceTo(sim3DEarth.position) > 1.35) {
                            const startPos = new THREE.Vector3(
                                (Math.random() - 0.5) * 1.2,
                                (Math.random() - 0.5) * 1.2,
                                (Math.random() - 0.5) * 1.2
                            ).normalize().multiplyScalar(1.0);
                            ray.position.copy(startPos);
                        }
                    });
                }

                sim3DRenderer.render(sim3DScene, sim3DCamera);
            }
            animate3DSim();
        }

        // UPDATE SIMULATION PHYSICS BASED ON SLIDERS
        function update3DSimulation() {
            const co2 = parseFloat(document.getElementById('slider-co2').value);
            const ch4 = parseFloat(document.getElementById('slider-ch4').value);
            const n2 = parseFloat(document.getElementById('slider-n2').value);
            const sun = parseFloat(document.getElementById('slider-sun').value);

            document.getElementById('val-co2').innerText = `${co2} ppm`;
            document.getElementById('val-ch4').innerText = `${ch4} ppm`;
            document.getElementById('val-n2').innerText = `${n2}% Atmosfer`;
            document.getElementById('val-sun').innerText = `${sun}% (Normal)`;

            const deltaCO2 = (co2 - 280) * 0.012;
            const deltaCH4 = (ch4 - 0.8) * 0.8;
            const deltaSun = (sun - 100) * 0.15;

            const temp = (14.0 + deltaCO2 + deltaCH4 + deltaSun).toFixed(1);
            const trappedPct = Math.min(95, Math.max(10, Math.round(20 + deltaCO2 * 3 + deltaCH4 * 5))).toFixed(0);

            document.getElementById('sim3d-temp-readout').innerText = `Suhu Model: ${temp}°C`;
            document.getElementById('sim3d-trapped-readout').innerText = `Panas Terperangkap: ${trappedPct}%`;

            if (sim3DAtmoMesh) {
                const opacityVal = Math.min(0.65, 0.15 + (co2 / 1000) * 0.4);
                sim3DAtmoMesh.material.opacity = opacityVal;
                if (temp > 18) sim3DAtmoMesh.material.color.setHex(0xef4444);
                else if (temp < 0) sim3DAtmoMesh.material.color.setHex(0x38bdf8);
                else sim3DAtmoMesh.material.color.setHex(0x10b981);
            }

            if (gasParticlesGroup) {
                while(gasParticlesGroup.children.length > 0) {
                    gasParticlesGroup.remove(gasParticlesGroup.children[0]);
                }

                const particleCount = Math.floor(co2 / 8);
                const geom = new THREE.SphereGeometry(0.035, 8, 8);
                const matCO2 = new THREE.MeshBasicMaterial({ color: 0x34d399 });
                const matCH4 = new THREE.MeshBasicMaterial({ color: 0x2dd4bf });

                for(let i=0; i<particleCount; i++) {
                    const particle = new THREE.Mesh(geom, (i % 5 === 0) ? matCH4 : matCO2);
                    const r = 1.05 + Math.random() * 0.22;
                    const theta = Math.random() * Math.PI * 2;
                    const phi = Math.acos((Math.random() * 2) - 1);

                    particle.position.set(
                        r * Math.sin(phi) * Math.cos(theta),
                        r * Math.sin(phi) * Math.sin(theta),
                        r * Math.cos(phi)
                    );
                    gasParticlesGroup.add(particle);
                }
            }

            if (solarRaysGroup && infraRaysGroup && sim3DSun) {
                while(solarRaysGroup.children.length > 0) solarRaysGroup.remove(solarRaysGroup.children[0]);
                while(infraRaysGroup.children.length > 0) infraRaysGroup.remove(infraRaysGroup.children[0]);

                const rayGeom = new THREE.SphereGeometry(0.03, 8, 8);
                const sunRayMat = new THREE.MeshBasicMaterial({ color: 0xfacc15 });
                const infraRayMat = new THREE.MeshBasicMaterial({ color: 0xef4444 });

                const numRays = Math.floor(sun / 8);
                for(let i=0; i<numRays; i++) {
                    const ray = new THREE.Mesh(rayGeom, sunRayMat);
                    ray.position.copy(sim3DSun.position);
                    const target = new THREE.Vector3(0,0,0);
                    const dir = target.sub(sim3DSun.position).normalize().multiplyScalar(0.045);

                    ray.userData = { vx: dir.x, vy: dir.y, vz: dir.z };
                    solarRaysGroup.add(ray);
                }

                const numInfra = Math.floor(trappedPct / 2.5);
                for(let i=0; i<numInfra; i++) {
                    const ray = new THREE.Mesh(rayGeom, infraRayMat);
                    const startPos = new THREE.Vector3(
                        (Math.random()-0.5)*1.2,
                        (Math.random()-0.5)*1.2,
                        (Math.random()-0.5)*1.2
                    ).normalize().multiplyScalar(1.0);
                    ray.position.copy(startPos);

                    const dir = startPos.clone().normalize().multiplyScalar(0.025);
                    ray.userData = { vx: dir.x, vy: dir.y, vz: dir.z };
                    infraRaysGroup.add(ray);
                }
            }
        }

        // GLOSARIUM FILTER
        function filterGlosarium() {
            const query = document.getElementById('glosarium-search').value.toLowerCase();
            const cards = document.querySelectorAll('.term-card');
            cards.forEach(card => {
                const text = card.innerText.toLowerCase();
                card.style.display = text.includes(query) ? 'block' : 'none';
            });
        }
